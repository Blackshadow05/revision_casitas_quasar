<template>
  <q-page class="q-pa-md">
    <div class="q-mb-md">
      <q-btn 
        label="Sincronizar Datos" 
        color="primary" 
        @click="cargarDatos" 
        :loading="loading"
      />
    </div>

    <!-- Ejemplo de visualización -->
    <div v-if="datos">
      <!-- SECCIÓN CHECK-OUTS -->
      <q-card class="q-mb-md" v-if="datos.checkouts.length">
        <q-card-section class="bg-red-5 text-white">
          <div class="text-h6">Check-Outs (Salidas)</div>
        </q-card-section>
        <q-separator />
        <q-list separator>
          <q-item v-for="(item, index) in datos.checkouts" :key="'out-'+index">
            <q-item-section avatar>
              <q-avatar color="red" text-color="white" size="sm">{{ item.casa }}</q-avatar>
            </q-item-section>
            <q-item-section>
              <q-item-label>{{ item.huesped }}</q-item-label>
              <q-item-label caption>Hora: {{ item.hora }}</q-item-label>
            </q-item-section>
          </q-item>
        </q-list>
      </q-card>

      <!-- SECCIÓN CHECK-INS -->
      <q-card class="q-mb-md" v-if="datos.checkins.length">
        <q-card-section class="bg-green-5 text-white">
          <div class="text-h6">Check-Ins (Llegadas)</div>
        </q-card-section>
        <q-separator />
        <q-list separator>
          <q-item v-for="(item, index) in datos.checkins" :key="'in-'+index">
            <q-item-section avatar>
              <q-avatar color="green" text-color="white" size="sm">{{ item.casa }}</q-avatar>
            </q-item-section>
            <q-item-section>
              <q-item-label>{{ item.huesped }}</q-item-label>
              <q-item-label caption>ETA: {{ item.hora }}</q-item-label>
            </q-item-section>
          </q-item>
        </q-list>
      </q-card>

      <!-- SECCIÓN EXPERIENCIAS -->
      <q-card class="q-mb-md" v-if="datos.experiencias.length">
        <q-card-section class="bg-blue-5 text-white">
          <div class="text-h6">Experiencias Confirmadas</div>
        </q-card-section>
        <q-separator />
        <q-list separator>
          <q-item v-for="(item, index) in datos.experiencias" :key="'exp-'+index">
            <q-item-section>
              <q-item-label overline>{{ item.hora }} - Casa {{ item.casita }}</q-item-label>
              <q-item-label>{{ item.tour }}</q-item-label>
              <q-item-label caption>{{ item.guia }}</q-item-label>
            </q-item-section>
          </q-item>
        </q-list>
      </q-card>
    </div>
  </q-page>
</template>

<script setup>
import { ref } from 'vue';
import { useQuasar } from 'quasar';

const $q = useQuasar();
const loading = ref(false);
const datos = ref(null); // Aquí se guardarán los datos

const URL_SCRIPT = 'https://script.google.com/macros/s/AKfycbwvfYpNUOlPW0LKnymGTw6zOU_CW3XbRnu6L1aVjj6Mo145S9kIhVj5OiLXKgGbH6F5/exec'; // PEGA TU URL AQUÍ

const cargarDatos = async () => {
  loading.value = true;
  try {
    const response = await fetch(URL_SCRIPT);
    const result = await response.json();
    
    if (result.error) {
      $q.notify({ type: 'negative', message: result.error });
    } else {
      datos.value = result;
      $q.notify({ type: 'positive', message: 'Datos actualizados' });
    }
  } catch (e) {
    $q.notify({ type: 'negative', message: 'Error de conexión' });
  } finally {
    loading.value = false;
  }
};
</script>
